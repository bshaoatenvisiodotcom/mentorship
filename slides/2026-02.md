# <!-- fit --> Clipboard

---

## <!-- fit --> `document.execCommand()`

---

- High compatibility
- `copy` / `cut` / `paste`

---
```js
const from = document.querySelector("#from");
from.select();
document.execCommand("copy");
```

```js
const to = document.querySelector("#to");
to.focus();
document.execCommand("paste");
```

---

## Limits

- Only works with selected content
- Synchronous; can block on large payloads
- Modern replacement: Async Clipboard API

---

## <!-- fit --> Async Clipboard API

---

- Entry point: `navigator.clipboard`
- Promise-based async API, non-blocking UI
- Supports text and binary data (e.g., images)

---

```js
const clipboard = navigator.clipboard;
```

---

## Security & Permissions

- HTTPS only (`localhost` is allowed)
- `clipboard-write`: usually granted automatically
- `clipboard-read`: requires explicit user consent

---

## `readText()`

```js
document.body.addEventListener("click", async (e) => {
  const text = await navigator.clipboard.readText();
  console.log(text);
});
```

```js
async function getClipboardContents() {
  try {
    const text = await navigator.clipboard.readText();
    console.log("Pasted content: ", text);
  } catch (err) {
    console.error("Failed to read clipboard contents: ", err);
  }
}
```

---

## `read()`

```js
async function getClipboardContents() {
  try {
    const clipboardItems = await navigator.clipboard.read();
    for (const clipboardItem of clipboardItems) {
      for (const type of clipboardItem.types) {
        const blob = await clipboardItem.getType(type);
        console.log(URL.createObjectURL(blob));
      }
    }
  } catch (err) {
    console.error(err.name, err.message);
  }
}
```

---

## `writeText()`

```js
document.body.addEventListener("click", async (e) => {
  await navigator.clipboard.writeText("Yo");
});
```

```js
async function copyPageUrl() {
  try {
    await navigator.clipboard.writeText(location.href);
    console.log("Page URL copied to clipboard");
  } catch (err) {
    console.error("Failed to copy: ", err);
  }
}
```

##  <!-- fit --> `copy` / `cut`

---

```js
const source = document.querySelector(".source");

source.addEventListener("copy", (event) => {
  const selection = document.getSelection();
  event.clipboardData.setData("text/plain", selection.toString().toUpperCase());
  event.preventDefault();
});
```

- `setData(type, data)`
- `getData(type)`
- `clearData([type])`
- `items`

---

```js
const clipboardItems = [];

document.addEventListener("copy", async (e) => {
  e.preventDefault();
  try {
    let clipboardItems = [];
    for (const item of e.clipboardData.items) {
      if (!item.type.startsWith("image/")) {
        continue;
      }
      clipboardItems.push(
        new ClipboardItem({
          [item.type]: item,
        }),
      );
      await navigator.clipboard.write(clipboardItems);
      console.log("Image copied.");
    }
  } catch (err) {
    console.error(err.name, err.message);
  }
});
```

## <!--fit--> `paste`
---

document.addEventListener("paste", async (e) => {
